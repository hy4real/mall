Task 4.2 — 重构 relateAndInsertList

PmsProductServiceImpl.java:252-267
现状：用 Method.invoke() 反射实现泛型批量插入，运行时可能 NPE/类型错误，不可测试
// 重构方案：定义泛型接口
public interface BatchInsertDao<T> {
    void insertList(List<T> list);
}

// 定义 ProductRelation 接口让所有关联实体实现
public interface ProductRelation {
    void setId(Long id);
    void setProductId(Long productId);
}

// 重构后的方法：类型安全，可测试
private <T extends ProductRelation> void relateAndInsert(
        BatchInsertDao<T> dao, List<T> list, Long productId) {
    if (list == null || list.isEmpty()) return;
    list.forEach(item -> { item.setId(null); item.setProductId(productId); });
    dao.insertList(list);
}

@RestController
@RequestMapping("/ai")
@RequiredArgsConstructor
public class AiController {
    private final AiChatService chatService;

    @PostMapping("/chat")
    public CommonResult<String> chat(@RequestBody ChatRequest req) {
        return CommonResult.success(chatService.answer(req));
    }

    @GetMapping(value = "/chat/stream", produces = "text/event-stream")
    public Flux<String> chatStream(@RequestParam String question) {
        return chatService.streamAnswer(question);
    }
}
